"use strict";
const http = require("http");
const fs = require("fs");
const path = require("path");
const { DatabaseSync } = require("node:sqlite");
const { ZteF672YDriver } = require("./zteDriver");

const PORT = Number(process.env.SWC_PORT || 8765);
const PUBLIC_DIR = path.join(__dirname, "..");
const MIME = { ".html": "text/html; charset=utf-8", ".js": "application/javascript; charset=utf-8", ".json": "application/json", ".png": "image/png", ".jpg": "image/jpeg", ".svg": "image/svg+xml", ".css": "text/css" };
const SWC_DIR = __dirname;
const DB_PATH = path.join(SWC_DIR, "data", "swc.db");
const TG_CFG_PATH = path.join(SWC_DIR, "..", "scanner", "telegramConfig.json");

const db = new DatabaseSync(DB_PATH);
const router = new ZteF672YDriver();
const sseClients = new Set();

let state = {
  routerOnline: false,
  lastScanAt: 0,
  lastError: "",
  autoAlertNewDevice: true
};

// Migrasi/ensure schema
db.exec(`
  CREATE TABLE IF NOT EXISTS devices (
    mac TEXT PRIMARY KEY,
    ip TEXT,
    hostname TEXT,
    alias TEXT,
    conn_type TEXT,
    status TEXT DEFAULT "online",
    block_mode TEXT DEFAULT "none",
    is_known INTEGER DEFAULT 0,
    first_seen INTEGER,
    last_seen INTEGER
  );
  CREATE TABLE IF NOT EXISTS event_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    mac TEXT,
    event_type TEXT,
    detail TEXT,
    created_at INTEGER
  );
  CREATE TABLE IF NOT EXISTS schedules (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    mac TEXT,
    action TEXT,
    start_time TEXT,
    end_time TEXT,
    enabled INTEGER DEFAULT 1
  );
`);

async function sendTelegram(text) {
  try {
    if (!fs.existsSync(TG_CFG_PATH)) return;
    const cfg = JSON.parse(fs.readFileSync(TG_CFG_PATH, "utf8"));
    if (!cfg.token || !cfg.chatId) return;
    await fetch(`https://api.telegram.org/bot${cfg.token}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        chat_id: cfg.chatId,
        text,
        parse_mode: "HTML"
      }),
      signal: AbortSignal.timeout(6000)
    });
  } catch (e) {
    console.error("[TG Err]", e.message);
  }
}

function logEvent(mac, eventType, detail) {
  db.prepare("INSERT INTO event_logs (mac, event_type, detail, created_at) VALUES (?, ?, ?, ?)")
    .run(mac || "-", eventType, detail, Date.now());
  db.prepare("DELETE FROM event_logs WHERE id NOT IN (SELECT id FROM event_logs ORDER BY id DESC LIMIT 150)").run();
}

function getSnapshot() {
  const devices = db.prepare("SELECT * FROM devices ORDER BY status DESC, last_seen DESC").all();
  const logs = db.prepare("SELECT * FROM event_logs ORDER BY id DESC LIMIT 30").all();
  const schedules = db.prepare("SELECT * FROM schedules ORDER BY id DESC").all();
  return { state, devices, logs, schedules };
}

function broadcast() {
  const payload = `data: ${JSON.stringify(getSnapshot())}\n\n`;
  for (const res of sseClients) {
    try { res.write(payload); } catch (_) { sseClients.delete(res); }
  }
}

let blockSyncQueue = Promise.resolve();
function applyRouterBlocks() {
  const task = async () => {
    const blocked = db.prepare("SELECT mac FROM devices WHERE block_mode = 'blocked'").all().map(r => r.mac);
    await router.syncBlockedMacs(blocked);
  };
  const next = blockSyncQueue.then(task, task);
  blockSyncQueue = next.catch(() => {});
  return next;
}

async function pollDevices() {
  try {
    const active = await router.getConnectedDevices();
    const now = Date.now();
    const activeSet = new Set(active.map(d => d.mac));

    for (const dev of active) {
      const existing = db.prepare("SELECT * FROM devices WHERE mac = ?").get(dev.mac);
      if (!existing) {
        db.prepare(`
          INSERT INTO devices (mac, ip, hostname, alias, conn_type, status, block_mode, is_known, first_seen, last_seen)
          VALUES (?, ?, ?, ?, ?, 'online', 'none', 0, ?, ?)
        `).run(dev.mac, dev.ip, dev.hostname, dev.hostname, dev.connType, now, now);

        logEvent(dev.mac, "NEW_DEVICE", `Perangkat baru terhubung: ${dev.hostname} (${dev.ip})`);
        if (state.autoAlertNewDevice) {
          sendTelegram(
            `🛡️ <b>SWC Wi-Fi Alert — Perangkat Baru</b>\n` +
            `• Nama: <code>${dev.hostname}</code>\n` +
            `• IP: <code>${dev.ip}</code>\n` +
            `• MAC: <code>${dev.mac}</code>`
          );
        }
      } else {
        db.prepare(`
          UPDATE devices SET ip = ?, hostname = ?, conn_type = ?, status = 'online', last_seen = ?
          WHERE mac = ?
        `).run(dev.ip, dev.hostname, dev.connType, now, dev.mac);
      }
    }

    const allStored = db.prepare("SELECT mac, status FROM devices").all();
    for (const row of allStored) {
      if (!activeSet.has(row.mac) && row.status === "online") {
        db.prepare("UPDATE devices SET status = 'offline' WHERE mac = ?").run(row.mac);
      }
    }

    state.routerOnline = true;
    state.lastScanAt = now;
    state.lastError = "";
  } catch (e) {
    state.routerOnline = false;
    state.lastError = e.message;
  }
  broadcast();
}

function checkSchedules() {
  const now = new Date();
  const hhmm = now.toTimeString().slice(0, 5);
  const rows = db.prepare("SELECT * FROM schedules WHERE enabled = 1").all();
  let changed = false;

  for (const s of rows) {
    const dev = db.prepare("SELECT * FROM devices WHERE mac = ?").get(s.mac);
    if (!dev) continue;
    const inWindow = s.start_time <= s.end_time
      ? (hhmm >= s.start_time && hhmm < s.end_time)
      : (hhmm >= s.start_time || hhmm < s.end_time);

    const targetMode = inWindow ? "blocked" : "none";
    if (dev.block_mode !== targetMode) {
      db.prepare("UPDATE devices SET block_mode = ? WHERE mac = ?").run(targetMode, s.mac);
      logEvent(s.mac, "SCHEDULE", `Jadwal otomatis: ${dev.alias || dev.hostname} -> ${targetMode === "blocked" ? "BLOKIR" : "IZINKAN"}`);
      changed = true;
    }
  }
  if (changed) {
    applyRouterBlocks().then(broadcast).catch(err => {
      state.lastError = err.message;
      broadcast();
    });
  }
}

function readJsonBody(req) {
  return new Promise((resolve, reject) => {
    let raw = "";
    req.on("data", chunk => { raw += chunk; if (raw.length > 65536) req.destroy(); });
    req.on("end", () => {
      try { resolve(raw ? JSON.parse(raw) : {}); } catch (e) { reject(e); }
    });
  });
}

const server = http.createServer(async (req, res) => {
  res.setHeader("Access-Control-Allow-Origin", "*");
  res.setHeader("Access-Control-Allow-Methods", "GET,POST,DELETE,OPTIONS");
  res.setHeader("Access-Control-Allow-Headers", "Content-Type");
  res.setHeader("Access-Control-Allow-Private-Network", "true");
  if (req.method === "OPTIONS") { res.writeHead(204); return res.end(); }

  const url = new URL(req.url, `http://127.0.0.1:${PORT}`);
  const sendJson = (code, obj) => {
    res.writeHead(code, { "Content-Type": "application/json; charset=utf-8" });
    res.end(JSON.stringify(obj));
  };

  try {
    if (req.method === "GET" && url.pathname === "/api/stream") {
      res.writeHead(200, {
        "Content-Type": "text/event-stream",
        "Cache-Control": "no-cache",
        "Connection": "keep-alive"
      });
      res.write(`data: ${JSON.stringify(getSnapshot())}\n\n`);
      sseClients.add(res);
      req.on("close", () => sseClients.delete(res));
      return;
    }

    if (req.method === "GET" && url.pathname === "/api/state") {
      return sendJson(200, getSnapshot());
    }

    if (req.method === "POST" && url.pathname === "/api/scan") {
      await pollDevices();
      return sendJson(200, { ok: true, ...getSnapshot() });
    }

    if (req.method === "POST" && url.pathname === "/api/device/block") {
      const { mac, block } = await readJsonBody(req);
      if (!mac) return sendJson(400, { error: "MAC wajib diisi" });
      const normMac = mac.toLowerCase();
      const mode = block ? "blocked" : "none";
      const dev = db.prepare("SELECT * FROM devices WHERE mac = ?").get(normMac);
      const prevMode = dev ? dev.block_mode : "none";
      const label = dev ? (dev.alias || dev.hostname) : normMac;
      db.prepare("UPDATE devices SET block_mode = ? WHERE mac = ?").run(mode, normMac);

      try {
        await applyRouterBlocks();
      } catch (syncErr) {
        db.prepare("UPDATE devices SET block_mode = ? WHERE mac = ?").run(prevMode, normMac);
        throw syncErr;
      }
      logEvent(normMac, block ? "BLOCK" : "UNBLOCK", `${block ? "Memblokir" : "Membuka blokir"} ${label}`);
      sendTelegram(
        `${block ? "🚫" : "✅"} <b>SWC Wi-Fi — ${block ? "Perangkat Diblokir" : "Blokir Dibuka"}</b>\n` +
        `• Perangkat: <b>${label}</b>\n` +
        `• MAC: <code>${normMac}</code>`
      );
      broadcast();
      return sendJson(200, { ok: true, ...getSnapshot() });
    }

    if (req.method === "POST" && url.pathname === "/api/device/meta") {
      const { mac, alias, is_known } = await readJsonBody(req);
      if (!mac) return sendJson(400, { error: "MAC wajib diisi" });
      const normMac = mac.toLowerCase();
      if (typeof alias === "string") {
        db.prepare("UPDATE devices SET alias = ? WHERE mac = ?").run(alias.trim(), normMac);
      }
      if (typeof is_known !== "undefined") {
        db.prepare("UPDATE devices SET is_known = ? WHERE mac = ?").run(is_known ? 1 : 0, normMac);
      }
      logEvent(normMac, "UPDATE", `Memperbarui profil perangkat ${normMac}`);
      broadcast();
      return sendJson(200, { ok: true, ...getSnapshot() });
    }

    if (req.method === "POST" && url.pathname === "/api/schedule") {
      const { mac, start_time, end_time } = await readJsonBody(req);
      if (!mac || !start_time || !end_time) return sendJson(400, { error: "Data jadwal tidak lengkap" });
      db.prepare("INSERT INTO schedules (mac, action, start_time, end_time, enabled) VALUES (?, 'block', ?, ?, 1)")
        .run(mac.toLowerCase(), start_time, end_time);
      logEvent(mac.toLowerCase(), "SCHEDULE_ADD", `Jadwal blokir ${start_time} - ${end_time}`);
      checkSchedules();
      broadcast();
      return sendJson(200, { ok: true, ...getSnapshot() });
    }

    if (req.method === "DELETE" && url.pathname === "/api/schedule") {
      const id = Number(url.searchParams.get("id"));
      db.prepare("DELETE FROM schedules WHERE id = ?").run(id);
      broadcast();
      return sendJson(200, { ok: true, ...getSnapshot() });
    }

    const relPath = url.pathname === "/" ? "/index.html" : (url.pathname === "/wifi" ? "/swc/public.html" : url.pathname);
    const filePath = path.normalize(path.join(PUBLIC_DIR, relPath));
    if (filePath.startsWith(PUBLIC_DIR) && fs.existsSync(filePath) && fs.statSync(filePath).isFile()) {
      const ext = path.extname(filePath).toLowerCase();
      res.writeHead(200, { "Content-Type": MIME[ext] || "application/octet-stream", "Cache-Control": "no-store, no-cache, must-revalidate" });
      return fs.createReadStream(filePath).pipe(res);
    }
    sendJson(404, { error: "Endpoint tidak ditemukan" });
  } catch (e) {
    sendJson(500, { error: e.message });
  }
});

"use strict";
const crypto = require("crypto");

class ZteF672YDriver {
  constructor(opts = {}) {
    this.base = opts.base || process.env.ZTE_URL || "http://192.168.1.1";
    this.user = opts.user || process.env.ZTE_USER || "admin";
    this.pass = opts.pass || process.env.ZTE_PASS || "Web@0063";
    this.ck = {};
    this.pubKey = "";
    this.tok = "";
    this.lock = Promise.resolve();
  }

  _upCk(res) {
    const list = res.headers.getSetCookie ? res.headers.getSetCookie() : [];
    for (const c of list) {
      const [pair] = c.split(";");
      const i = pair.indexOf("=");
      if (i > 0) this.ck[pair.slice(0, i).trim()] = pair.slice(i + 1).trim();
    }
  }

  _signCheck(payloadStr) {
    if (!this.pubKey) return "";
    const digest = crypto.createHash("sha256").update(payloadStr).digest("hex");
    return crypto.publicEncrypt(
      { key: this.pubKey, padding: crypto.constants.RSA_PKCS1_PADDING },
      Buffer.from(digest)
    ).toString("base64");
  }

  _assertXmlOk(xml, context = "Router") {
    const mErr = String(xml || "").match(/<IF_ERRORSTR>([^<]+)<\/IF_ERRORSTR>/);
    if (mErr && mErr[1].trim() !== "SUCC") {
      throw new Error(`${context}: ${mErr[1].trim()}`);
    }
    if (String(xml || "").includes("Frm_Username")) {
      throw new Error(`${context}: SessionTimeout`);
    }
  }

  async _req(path, body = null, { ajax = true, signStr = null } = {}) {
    const headers = {
      "User-Agent": "Mozilla/5.0 (Linux; Android 17)",
      "Referer": this.base + "/"
    };
    if (ajax) headers["X-Requested-With"] = "XMLHttpRequest";
    const cookieStr = Object.entries(this.ck).map(([k, v]) => `${k}=${v}`).join("; ");
    if (cookieStr) headers["Cookie"] = cookieStr;
    if (body !== null) headers["Content-Type"] = "application/x-www-form-urlencoded; charset=UTF-8";
    if (signStr !== null && this.pubKey) headers["Check"] = this._signCheck(signStr);

    const res = await fetch(this.base + path, {
      method: body !== null ? "POST" : "GET",
      headers,
      body,
      signal: AbortSignal.timeout(9000)
    });
    this._upCk(res);
    const txt = await res.text();
    const mTok = txt.match(/_sessionTmpToken\s*=\s*"([^"]+)"/);
    if (mTok) {
      this.tok = mTok[1].replace(/\\x([0-9a-fA-F]{2})/g, (_, h) => String.fromCharCode(parseInt(h, 16)));
    }
    return txt;
  }

  async _runSingleSession(fn) {
    this.ck = {};
    this.tok = "";
    try {
      await this._req("/", null, { ajax: false });
      const initJson = JSON.parse(await this._req("/?_type=loginData&_tag=login_entry"));
      if (initJson.lockingTime && Number(initJson.lockingTime) > 0) {
        throw new Error(`Router terkunci sementara (${initJson.lockingTime} detik)`);
      }
      const saltXml = await this._req("/?_type=loginData&_tag=login_token");
      const mSalt = saltXml.match(/>([^<]+)</);
      if (!mSalt) throw new Error("Gagal membaca login_token router");
      const shaPass = crypto.createHash("sha256").update(this.pass + mSalt[1].trim()).digest("hex");

      const loginRes = JSON.parse(await this._req(
        "/?_type=loginData&_tag=login_entry",
        `action=login&Password=${shaPass}&Username=${encodeURIComponent(this.user)}&_sessionTOKEN=${encodeURIComponent(initJson.sess_token)}`
      ));
      if (!loginRes.login_need_refresh) {
        throw new Error(loginRes.loginErrMsg || "Login router ditolak");
      }

      const dash = await this._req("/", null, { ajax: false });
      if (dash.includes("Frm_Username")) {
        throw new Error("SessionTimeout saat memuat dashboard router");
      }
      const mTok = dash.match(/_sessionTmpToken\s*=\s*"([^"]+)"/);
      this.tok = mTok
        ? mTok[1].replace(/\\x([0-9a-fA-F]{2})/g, (_, h) => String.fromCharCode(parseInt(h, 16)))
        : loginRes.sess_token;
      const mPk = dash.match(/(-----BEGIN PUBLIC KEY-----[\s\S]*?-----END PUBLIC KEY-----)/);
      if (mPk) this.pubKey = mPk[1].replace(/\\n/g, "\n");

      return await fn();
    } finally {
      if (this.tok) {
        await this._req(
          "/?_type=loginData&_tag=logout_entry",
          `IF_LogOff=1&_sessionTOKEN=${encodeURIComponent(this.tok)}`
        ).catch(() => {});
        this.tok = "";
      }
    }
  }

  async _withSession(fn) {
    const runWithRetry = async () => {
      let lastErr;
      for (let attempt = 1; attempt <= 3; attempt++) {
        try {
          return await this._runSingleSession(fn);
        } catch (err) {
          lastErr = err;
          if (attempt < 3) {
            await new Promise(r => setTimeout(r, 900 * attempt));

        this._assertXmlOk(polApplyRes, "Terapkan ACLPolicy");
      }

      const verifyRuleXml = await this._req(
        `/?_type=menuData&_tag=wlan_macfilterrule_lua.lua&_sessionTOKEN=${encodeURIComponent(this.tok)}`,

        this._parseInstances(verifyRuleXml).map(r => (r.MACAddress || "").toLowerCase())
      );
      for (const mac of targetSet) {
        if (!savedMacs.has(mac)) {
          throw new Error(`Verifikasi gagal: MAC ${mac} belum masuk ke tabel ACL router`);
        }
      }

      return { synced: targetSet.size, policy: desiredPolicy };
    });

"bear" : "neutral"; push("macd", "MACD (12,26,9)", fmt(m.histogram, p < 1 ? 6 : 4), d, d === "bull" ? "Histogram positif" : d === "bear" ? "Histogram negatif" : "Mendekati nol"); }

  const e9 = TA.ema(c, 9), e21 = TA.ema(c, 21); ctx.ema21 = e21;
  if (!finite(e9) || !finite(e21)) na("ema", "EMA 9/21", "butuh 21 candle");
  else { const d = e9 > e21 && p > e21 ? "bull" : e9 < e21 && p < e21 ? "bear" : "neutral"; push("ema", "EMA 9/21", fmtP(e9) + " / " + fmtP(e21), d, d === "bull" ? "EMA 9 > EMA 21 & Harga > EMA 21" : d === "bear" ? "EMA 9 < EMA 21 & Harga < EMA 21" : "Persilangan netral"); }

  const s20 = TA.sma(c, 20), s50 = TA.sma(c, 50);
  if (!finite(s20) || !finite(s50)) na("sma", "SMA 20/50", "butuh 50 candle");
  else { const d = s20 > s50 && p > s50 ? "bull" : s20 < s50 && p < s50 ? "bear" : "neutral"; push("sma", "SMA 20/50", fmtP(s20) + " / " + fmtP(s50), d, d === "bull" ? "SMA 20 > SMA 50" : d === "bear" ? "SMA 20 < SMA 50" : "Netral"); }

  const b = TA.bb(c);
  if (!finite(b.upper) || !finite(b.lower) || b.upper <= b.lower) na("bb", "Bollinger Bands (20,2)", "butuh 20 candle");
  else { const pct = (p - b.lower) / (b.upper - b.lower); const d = (pct > 1 || pct < 0) ? "neutral" : pct >= 0.55 ? "bull" : pct <= 0.45 ? "bear" : "neutral"; push("bb", "Bollinger Bands (20,2)", "%B " + fmt(pct * 100, 0) + "%", d, d === "bull" ? "Di paruh atas band" : d === "bear" ? "Di paruh bawah band" : "Di tengah / luar band"); }

  const sr = TA.stochRsi(c);
  if (!finite(sr.k) || !finite(sr.d)) na("stochrsi", "Stochastic RSI", "butuh 35 candle");
  else { const d = sr.k > sr.d && sr.k < 80 ? "bull" : sr.k < sr.d && sr.k > 20 ? "bear" : "neutral"; push("stochrsi", "Stochastic RSI (14,14,3,3)", "K " + fmt(sr.k, 1) + " / D " + fmt(sr.d, 1), d, d === "bull" ? "%K > %D" : d === "bear" ? "%K < %D" : "Jenuh / Netral"); }

  const a = TA.adx(h, l, c); ctx.adx = a.adx;
  if (!finite(a.adx)) na("adx", "ADX (14)", "butuh 30 candle");
  else { const d = a.adx >= 20 && a.plusDI > a.minusDI ? "bull" : a.adx >= 20 && a.minusDI > a.plusDI ? "bear" : "neutral"; push("adx", "ADX (14)", fmt(a.adx, 1) + " (+DI " + fmt(a.plusDI, 1) + " / -DI " + fmt(a.minusDI, 1) + ")", d, d === "bull" ? "ADX ≥ 20 (+DI > -DI)" : d === "bear" ? "ADX ≥ 20 (-DI > +DI)" : "ADX < 20"); }

  const cci = TA.cci(h, l, c);
  if (!finite(cci)) na("cci", "CCI (20)", "butuh 20 candle");
  else { const d = cci > 100 && cci <= 200 ? "bull" : cci < -100 && cci >= -200 ? "bear" : "neutral"; push("cci", "CCI (20)", fmt(cci, 1), d, d === "bull" ? "CCI > +100" : d === "bear" ? "CCI < -100" : "Netral"); }

  const obv = TA.obvSeries(cs);
  if (n < 26) na("obv", "OBV", "butuh 26 candle");
  else { const oe = TA.emaSeries(obv, 20), eNow = oe[n - 1], now = obv[n - 1], prev = obv[n - 6]; if (!finite(eNow)) na("obv", "OBV", "EMA OBV N/A"); else { const d = now > eNow && now > prev ? "bull" : now < eNow && now < prev ? "bear" : "neutral"; push("obv", "OBV", fmt(now, 0), d, d === "bull" ? "OBV naik di atas EMA 20" : d === "bear" ? "OBV turun di bawah EMA 20" : "OBV netral"); } }

  const atrVals = atrS.slice(-20).filter(finite);
  if (!finite(atrNow) || atrVals.length < 10 || n < 5) na("atr", "ATR (14)", "butuh 25 candle");
  else { const avg = mean(atrVals), exp = atrNow > avg * 1.05, chg = c[n - 1] - c[n - 4]; ctx.atrExp = exp; const d = exp && Math.abs(chg) > 0.5 * atrNow ? (chg > 0 ? "bull" : "bear") : "neutral"; push("atr", "ATR (14)", fmtP(atrNow), d, d !== "neutral" ? "Volatilitas meluas" : "Volatilitas normal"); }

  return { list, ctx };
}

function swingExt(cs, k, type) { const s = cs.slice(-k); return type === "L" ? Math.min(...s.map(x => x.low)) : Math.max(...s.map(x => x.high)); }
function buildLevels(s, cs, ctx) {
  const p = cs[cs.length - 1].close, atr = ctx.atr, ema21 = ctx.ema21;
  let far = p - s * 0.5 * atr;
  if (finite(ema21) && ((s === 1 && ema21 < p && ema21 > far) || (s === -1 && ema21 > p && ema21 < far))) far = ema21;
  if (Math.abs(p - far) < 0.1 * atr) far = p - s * 0.1 * atr;
  const low = Math.min(p, far), high = Math.max(p, far), mid = (low + high) / 2;
  let sl;
  if (s === 1) { const sw = swingExt(cs, 10, "L") - 0.25 * atr; sl = sw < low ? sw : mid - 1.5 * atr; sl = Math.max(mid - 2.5 * atr, Math.min(sl, mid - 1.0 * atr)); }
  else { const sw = swingExt(cs, 10, "H") + 0.25 * atr; sl = sw > high ? sw : mid + 1.5 * atr; sl = Math.min(mid + 2.5 * atr, Math.max(sl, mid + 1.0 * atr)); }
  const risk = Math.abs(mid - sl);
  return { low, high, mid, sl, risk, rr: [2, 3, 5], tps: [2, 3, 5].map(k => mid + s * k * risk) };
}

function computeSignal(all) {
  const cs = all.slice(-SIG.window), last = cs[cs.length - 1];
  const base = { time: last ? last.time : 0, price: last ? last.close : NaN, list: [], g: 0, y: 0, r: 0, nv: 0, strength: null, levels: null, filterMsg: "" };
  if (cs.length < SIG.minCandles) return { ...base, status: "INSUFFICIENT", reason: `Data candle belum cukup (${cs.length}/${SIG.minCandles}).`, invalid: "--" };
  const { list, ctx } = evaluateIndicators(cs), valid = list.filter(x => x.dir !== "na"), nv = valid.length;
  const g = valid.filter(x => x.dir === "bull").length, r = valid.filter(x => x.dir === "bear").length, y = valid.filter(x => x.dir === "neutral").length;
  const res = { ...base, list, g, y, r, nv };
  if (nv < SIG.minValid) return { ...res, status: "INSUFFICIENT", reason: `Hanya ${nv} indikator valid.`, invalid: "--" };

  const minVotes = Math.ceil(0.6 * nv), minMargin = Math.ceil(0.3 * nv);
  let status = "NEUTRAL";
  if (g >= minVotes && g - r >= minMargin) status = "BUY";
  else if (r >= minVotes && r - g >= minMargin) status = "SELL";

  const s = status === "BUY" ? 1 : status === "SELL" ? -1 : 0, adxV = ctx.adx;
  if (s !== 0) {
    const ema = list.find(x => x.key === "ema"), sma = list.find(x => x.key === "sma"), adx = list.find(x => x.key === "adx"), opp = s === 1 ? "bear" : "bull";
    if (ema && sma && ema.dir === opp && sma.dir === opp) res.filterMsg = "Filter tren bertentangan.";
    else if (adx && adx.dir === opp && adxV >= 25) res.filterMsg = "Filter ADX bertentangan.";
    if (res.filterMsg) status = "NEUTRAL";
  }
  res.status = status;
  if (status === "NEUTRAL") { res.reason = "Belum ada konfirmasi suara mayoritas indikator."; res.invalid = "--"; return res; }

  const sc = d => d === "bull" ? 1 : d === "bear" ? -1 : 0;
  let num = 0, den = 0;
  for (const gr in GROUP_W) { const mm = valid.filter(x => x.group === gr); if (!mm.length) continue; num += GROUP_W[gr] * s * mean(mm.map(x => sc(x.dir))); den += GROUP_W[gr]; }
  const dom = den ? clamp(num / den, -1, 1) : 0, adxF = finite(adxV) ? clamp((adxV - 15) / 25, 0, 1) : 0, volF = ctx.atrExp ? 1 : 0.4;
  res.strength = Math.round(100 * clamp(0.65 * Math.max(0, dom) + 0.2 * adxF + 0.15 * volF, 0, 1));
  res.levels = buildLevels(s, cs, ctx);
  res.reason = `${s === 1 ? "BUY" : "SELL"} terkonfirmasi: ${g} hijau, ${y} kuning, ${r} merah dari ${nv} indikator (Skor ${res.strength}/100).`;
  res.invalid = `Candle ditutup menembus Stop Loss ${fmtP(res.levels.sl)}.`;
  return res;
}

function runBacktest(cs) {
  const N = cs.length, start = SIG.backtestStart, trades = [];
  if (N < start + SIG.maxBars + 5) return null;
  let i = start;
  while (i < N - 1) {
    const s = computeSignal(cs.slice(0, i + 1));
    if ((s.status !== "BUY" && s.status !== "SELL") || !s.levels || !(s.levels.risk > 0)) { i++; continue; }
    const dir = s.status === "BUY" ? 1 : -1, entry = cs[i + 1].open, risk = s.levels.risk, sl = entry - dir * risk, tp1 = entry + dir * 2 * risk, last = i + SIG.maxBars;
    let res = null;
    for (let k = i + 1; k <= Math.min(last, N - 1); k++) {
      const c = cs[k], hitSL = dir === 1 ? c.low <= sl : c.high >= sl, hitTP = dir === 1 ? c.high >= tp1 : c.low <= tp1;
      if (hitSL) { res = { k, out: "SL" }; break; }
      if (hitTP) { res = { k, out: "TP1" }; break; }
      if (k === last) { res = { k, out: "TIMEOUT" }; break; }
    }
    if (!res) break;
    trades.push(res.out);
    i = res.k + 1;
  }
  const n = trades.length;
  if (!n) return { n: 0 };
  const wins = trades.filter(x => x === "TP1").length;
  return { n, winRate: (100 * wins) / n };
}

function scheduleBacktest() {
  clearTimeout(state.btTimer);
  const key = CONFIG.symbol + "|" + CONFIG.interval;
  state.btTimer = setTimeout(() => {
    try {
      const r = runBacktest(state.candles);
      if (key !== CONFIG.symbol + "|" + CONFIG.interval) return;
      state.bt = r || { insufficient: true };
      renderSignal();
    } catch (_) {
      state.bt = { error: true };
      renderSignal();
    }
  }, 60);
}

function clearSignalLines() { if (state.candleSeries) state.sigLines.forEach(l => { try { state.candleSeries.removePriceLine(l); } catch (_) {} }); state.sigLines = []; }
function drawSignalLines(s) {
  clearSignalLines();
  if (!state.candleSeries || !s || !s.levels || (s.status !== "BUY" && s.status !== "SELL")) return;
  const lv = s.levels, add = (price, color, title) => state.sigLines.push(state.candleSeries.createPriceLine({ price, color, lineWidth: 1, lineStyle: 2, axisLabelVisible: true, title }));
  add(lv.mid, "#38bdf8", "ENTRY");
  add(lv.sl, "#ef5350", "SL");
  lv.tps.forEach((t, i) => add(t, "#26a69a", "TP" + (i + 1)));
}

function updateSignal() {
  const cs = state.candles;
  if (!cs.length) { state.signal = null; renderSignal(); return; }
  const key = CONFIG.symbol + "|" + CONFIG.interval + "|" + cs[cs.length - 1].time;
  if (key !== state.sigKey) {
    state.sigKey = key;
    state.signal = computeSignal(cs);
    drawSignalLines(state.signal);
  }
  renderSignal();
}

async function fetchHistory() {
  const gen = ++state.generation;
  try {
    const { data } = await apiFetch(`/api/v3/klines?symbol=${encodeURIComponent(CONFIG.symbol)}&interval=${encodeURIComponent(CONFIG.interval)}&limit=${CONFIG.historyLimit}`);
    if (gen !== state.generation) return;
    const now = Date.now();
    state.candles = data.map(d => ({ time: +d[0], open: +d[1], high: +d[2], low: +d[3], close: +d[4], volume: +d[5], closeTime: +d[6] })).filter(c => c.closeTime < now);
    state.currentPrice = state.candles[state.candles.length - 1].close;
    state.liveCandle = null;
    clearSignalLines(); state.sigKey = ""; state.signal = null; state.bt = null;
    setChartData(state.candles, true);
    updateSignal();
    scheduleBacktest();
    EMA_EXP.init();
    connectWS();
  } catch (_) {
    setTimeout(fetchHistory, 3000);
  }
}

function connectWS() {
  if (state.ws) { try { state.ws.close(); } catch (_) {} }
  let ws;
  try { ws = new WebSocket(CONFIG.wsBases[0] + CONFIG.symbol.toLowerCase() + "@kline_" + CONFIG.interval); } catch (_) { setTimeout(fetchHistory, 3000); return; }
  state.ws = ws;
  ws.onopen = () => setStatus("Terhubung · " + CONFIG.symbol, "green");
  ws.onmessage = e => { try { const m = JSON.parse(e.data); if (m.k) processKline(m.k); } catch (_) {} };
  ws.onclose = () => setTimeout(fetchHistory, 3000);
}

function processKline(k) {
  const c = { time: +k.t, open: +k.o, high: +k.h, low: +k.l, close: +k.c, volume: +k.v, closeTime: +k.T, isClosed: !!k.x };
  state.currentPrice = c.close;
  state.liveCandle = c;
  $("sgPrice").textContent = "$" + fmtP(c.close);
  showLegend(c);
  if (state.candleSeries) {
    state.candleSeries.update({ time: toCT(c.time), open: c.open, high: c.high, low: c.low, close: c.close });
    state.volumeSeries.update({ time: toCT(c.time), value: c.volume, color: c.close >= c.open ? "rgba(38,166,154,.35)" : "rgba(239,83,80,.35)" });
  }
  if (!c.isClosed) return;
  const idx = state.candles.findIndex(x => x.time === c.time);
  if (idx >= 0) state.candles[idx] = c; else state.candles.push(c);
  state.liveCandle = null;
  updateSignal();
  scheduleBacktest();
  if (idx < 0) EMA_EXP.init();
}

function reconnectNow() {
  state.generation++;
  if (state.ws) { try { state.ws.close(); } catch (_) {} }
  state.candles = [];
  fetchHistory();
}

function radarFeatures(cs) {
  const n = cs.length;
  if (n < 36) return null;
  const c = cs.map(x => x.close), h = cs.map(x => x.high), l = cs.map(x => x.low), qv = cs.map(x => x.qv);
  const m20 = i => mean(qv.slice(i - 20, i)), recent = [];
  for (let i = n - 12; i < n; i++) { const b = m20(i); recent.push({ rv: b > 0 ? qv[i] / b : NaN, open: cs[i].open, time: cs[i].time }); }
  const rvol = recent[recent.length - 1].rv, bP = mean(qv.slice(n - 23, n - 3));
  const persist = [n - 3, n - 2, n - 1].filter(i => bP > 0 && qv[i] >= 1.5 * bP).length;
  const q3 = mean(qv.slice(n - 6, n - 3)), accel = q3 > 0 ? mean(qv.slice(n - 3)) / q3 : NaN;
  const atr = TA.atrSeries(h, l, c, 14)[n - 1];
  const ema21 = TA.ema(c, 21), rsi = TA.rsi(c), mac = TA.macd(c), m3 = atr > 0 ? (c[n - 1] - c[n - 4]) / atr : 0;
  let bu = 0, be = 0;
  if (m3 > 0.5) bu++; else if (m3 < -0.5) be++;
  if (c[n - 1] > ema21) bu++; else if (c[n - 1] < ema21) be++;
  if (rsi >= 55) bu++; else if (rsi <= 45) be++;
  if (mac.histogram > 0) bu++; else if (mac.histogram < 0) be++;
  const direction = bu >= 3 && be <= 1 ? "bull" : be >= 3 && bu <= 1 ? "bear" : "mixed";
  const chg = k => (n - 1 - k >= 0) ? (c[n - 1] / c[n - 1 - k] - 1) * 100 : NaN;
  return { rvol, accel, persist, recent, direction, chg5: chg(1), chg15: chg(3), chg60: chg(12), expire: cs[n - 1].closeTime + 302000 };
}

function radarClassify(sym) {
  const u = radar.uni.get(sym), f = radar.feat.get(sym);
  const o = { sym, price: u.price, q24: u.q24, status: "NO", score: 0, dist: NaN, start: null, rvol: NaN, accel: NaN, ch5: NaN, ch15: NaN, ch60: NaN };
  if (!f) return o;
  o.rvol = f.rvol; o.accel = f.accel; o.ch5 = f.chg5; o.ch15 = f.chg15; o.ch60 = f.chg60;
  const cand = f.recent.filter(r => finite(r.rv) && r.rv >= 2);
  const active = cand.length > 0 && (f.rvol >= 1.5 || f.persist >= 2);
  if (active) {
    o.start = cand[0];
    o.dist = ((u.price - o.start.open) / o.start.open) * 100;
    o.status = f.direction === "bull" ? "EB" : f.direction === "bear" ? "EE" : "VS";
  }
  o.score = finite(f.rvol) ? f.rvol * 15 + f.persist * 10 + (f.direction !== "mixed" ? 20 : 0) : 0;
  return o;
}
